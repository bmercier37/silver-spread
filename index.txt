<script>
function load() {
  const silverNY = 24.80;
  const silverLDN = 24.75;
  const silverSH = 26.10;
  const gold = 2050;

  document.getElementById("ny").innerText = silverNY.toFixed(2)+" $";
  document.getElementById("ldn").innerText = silverLDN.toFixed(2)+" $";
  document.getElementById("sh").innerText = silverSH.toFixed(2)+" $";

  const spread = silverSH - silverNY;
  const pct = (spread / silverNY) * 100;

  document.getElementById("spread").innerText =
    spread.toFixed(2)+" $ ("+pct.toFixed(2)+" %)";

  let alertTxt = "Marché normal";
  let alertClass = "ok";

  if (pct > 8) { alertTxt = "STRESS EXTRÊME"; alertClass="alert"; }
  else if (pct > 5) { alertTxt = "ALERTE TENSION"; alertClass="alert"; }
  else if (pct > 3) { alertTxt = "Tension physique"; alertClass="warn"; }

  document.getElementById("spreadAlert").innerHTML =
    `<span class="${alertClass}">${alertTxt}</span>`;

  // Backwardation / Contango
  document.getElementById("structure").innerText =
    silverSH > silverNY ? "Backwardation (physique tendu)" : "Contango";

  // Ratio Or / Argent
  const ratio = gold / silverNY;
  document.getElementById("ratio").innerText = ratio.toFixed(1);

  // Historique local
  let hist = JSON.parse(localStorage.getItem("spreadHist")) || [];
  hist.push(pct.toFixed(2));
  if (hist.length > 7) hist.shift();
  localStorage.setItem("spreadHist", JSON.stringify(hist));
  document.getElementById("history").innerText = hist.join(" % | ")+" %";

  // Stress global
  document.getElementById("stress").innerText =
    (pct > 5 && silverSH > silverNY)
    ? "⚠️ RISQUE DE SHORT SQUEEZE"
    : "Situation contrôlée";
}

load();
setInterval(load, 60000);
</script>
